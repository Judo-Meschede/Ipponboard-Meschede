set -euo pipefail
REPO="$HOME/Ipponboard-Meschede-Source"
URL="https://github.com/Judo-Meschede/Ipponboard-Meschede.git"

if [ -d "$REPO/.git" ]; then
  git -C "$REPO" fetch origin main
  git -C "$REPO" reset --hard origin/main
  git -C "$REPO" clean -fd
else
  rm -rf "$REPO"
  git clone --branch main --single-branch "$URL" "$REPO"
fi

echo
echo "GitHub-Stand geladen:"
echo "Version: $(cat "$REPO/CURRENT_VERSION.txt")"
git -C "$REPO" log -1 --oneline
